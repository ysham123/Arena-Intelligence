cmake_minimum_required(VERSION 3.24)
project(arena_intelligence VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(FetchContent)
FetchContent_Declare(json URL https://github.com/nlohmann/json/releases/download/v3.11.3/json.tar.xz DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
FetchContent_MakeAvailable(json)
find_package(Threads REQUIRED)
add_library(arena_core native/src/engine.cpp)
target_include_directories(arena_core PUBLIC native/include)
target_link_libraries(arena_core PUBLIC nlohmann_json::nlohmann_json Threads::Threads)
target_compile_options(arena_core PRIVATE -Wall -Wextra -Wpedantic)
add_executable(arena-sim native/src/main.cpp)
target_link_libraries(arena-sim PRIVATE arena_core)
target_compile_options(arena-sim PRIVATE -Wall -Wextra -Wpedantic)
include(CTest)
if(BUILD_TESTING)
  FetchContent_Declare(Catch2 URL https://github.com/catchorg/Catch2/archive/refs/tags/v3.8.1.tar.gz DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
  FetchContent_MakeAvailable(Catch2)
  add_executable(arena-native-tests native/tests/engine_tests.cpp)
  target_link_libraries(arena-native-tests PRIVATE arena_core Catch2::Catch2WithMain)
  add_test(NAME arena-native-tests COMMAND arena-native-tests)
  find_package(Python3 COMPONENTS Interpreter REQUIRED)
  add_test(NAME arena-native-transport COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/native/tests/test_transport.py $<TARGET_FILE:arena-sim>)
endif()
install(TARGETS arena-sim RUNTIME DESTINATION bin)
